class NFCCard(models.Model):
    """NFC 卡片對應表"""
    nfc_uid = models.CharField(max_length=50, unique=True, verbose_name="NFC UID")
    user = models.ForeignKey(User, on_delete=models.CASCADE, related_name='nfc_cards')
    card_name = models.CharField(max_length=100, default='主卡', verbose_name="卡片名稱")
    is_active = models.BooleanField(default=True, verbose_name="是否啟用")
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        db_table = 'nfc_cards'
        verbose_name = 'NFC卡片'
        verbose_name_plural = 'NFC卡片'
    
    def __str__(self):
        return f"{self.nfc_uid} - {self.user.username}"

from rest_framework.decorators import api_view
from rest_framework.response import Response
from rest_framework import status
from .models import NFCCard
from django.contrib.auth.models import User

@api_view(['GET'])
def nfc_lookup(request, nfc_uid):
    """
    根據 NFC UID 查詢使用者資訊
    
    URL: GET /api/nfc/{nfc_uid}/
    Example: GET /api/nfc/A1:B2:C3:D4/
    
    Response:
    {
        "user_id": 123,
        "username": "張大明",
        "first_name": "大明",
        "last_name": "張",
        "personnel_number": "EMP001",
        "groups": [
            {"group_id": 5, "group_name": "A車隊"}
        ]
    }
    """
    try:
        # 查詢 NFC 卡片
        nfc_card = NFCCard.objects.select_related('user').get(
            nfc_uid=nfc_uid,
            is_active=True
        )
        
        user = nfc_card.user
        profile = user.personnelprofile
        
        # 查詢使用者所屬的群組
        memberships = user.groupmember_set.select_related('group').all()
        groups = [
            {
                'group_id': m.group.id,
                'group_name': m.group.name,
                'role': m.role
            }
            for m in memberships
        ]
        
        return Response({
            'user_id': user.id,
            'username': user.username,
            'first_name': user.first_name,
            'last_name': user.last_name,
            'personnel_number': profile.personnel_number,
            'groups': groups
        }, status=status.HTTP_200_OK)
        
    except NFCCard.DoesNotExist:
        return Response({
            'error': '此 NFC 卡片尚未註冊或已停用'
        }, status=status.HTTP_404_NOT_FOUND)
    except Exception as e:
        return Response({
            'error': str(e)
        }, status=status.HTTP_500_INTERNAL_SERVER_ERROR)


@api_view(['PATCH'])
def update_trip_scores(request, trip_id):
    """
    更新行程評分
    
    URL: PATCH /api/trips/{trip_id}/scores/
    Body:
    {
        "score": 85.5,
        "in_car_score": 90.0,
        "out_car_score": 81.0,
        "ai_suggestion": "建議改善車距控制..."
    }
    """
    try:
        trip = Trip.objects.get(id=trip_id)
    except Trip.DoesNotExist:
        return Response({
            'error': '行程不存在'
        }, status=status.HTTP_404_NOT_FOUND)
    
    # 驗證評分範圍
    for field in ['score', 'in_car_score', 'out_car_score']:
        if field in request.data:
            value = request.data[field]
            if value < 0 or value > 100:
                return Response({
                    'error': f'{field} 必須在 0-100 之間'
                }, status=status.HTTP_400_BAD_REQUEST)
    
    # 更新資料
    if 'score' in request.data:
        trip.score = request.data['score']
    if 'in_car_score' in request.data:
        trip.in_car_score = request.data['in_car_score']
    if 'out_car_score' in request.data:
        trip.out_car_score = request.data['out_car_score']
    if 'ai_suggestion' in request.data:
        trip.ai_suggestion = request.data['ai_suggestion']
    
    trip.save()
    
    return Response({
        'id': trip.id,
        'score': trip.score,
        'in_car_score': trip.in_car_score,
        'out_car_score': trip.out_car_score,
        'ai_suggestion': trip.ai_suggestion
    }, status=status.HTTP_200_OK)
